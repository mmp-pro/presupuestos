function sincronizarConSheets() { if (articulos.length === 0) { alert('⚠️ No hay artículos para sincronizar'); return; } const datos = { articulos: articulos, fecha: new Date().toLocaleDateString('es-ES') }; const btn = event.target; const textoOriginal = btn.textContent; btn.textContent = '⏳ Sincronizando...'; btn.disabled = true; fetch(SCRIPT_URL, { method: 'POST', mode: 'no-cors', headers: { 'Content-Type': 'text/plain' }, body: JSON.stringify(datos) }) .then(() => { // Con no-cors, si llega aquí sin error, se envió correctamente setTimeout(() => { alert('✅ Datos sincronizados correctamente con Google Sheets'); limpiarTodo(); }, 1500); }) .catch(error => { console.error('Error completo:', error); alert('❌ Error al sincronizar. Verifica la consola para más detalles.'); }) .finally(() => { btn.textContent = textoOriginal; btn.disabled = false; }); }
